## 2. Spline Pulse Mathematics & Point-on-Curve Sampling

Given a parametric 3D curve $\vec{C}(t)$ where $t \in [0, 1]$ generated by `THREE.CatmullRomCurve3`:

$$\vec{P}_{\text{photon}}(t) = \vec{C}(t)$$

$$\vec{T}_{\text{tangent}}(t) = \frac{d\vec{C}(t)}{dt} \cdot \frac{1}{\|d\vec{C}(t)/dt\|}$$

To create an animated pulse with a fading tail of length $\Delta t$:
- Head point: $t_{\text{head}} = t$
- Tail point: $t_{\text{tail}} = \max(0, t - \Delta t)$

```typescript
import * as THREE from 'three';
import { animate } from 'animejs';

export class SplinePhoton {
  public mesh: THREE.Mesh;
  public progress = 0; // 0.0 -> 1.0

  constructor(
    public curve: THREE.CatmullRomCurve3,
    colorHex = 0xF59E0B // Cadmium Amber
  ) {
    const geometry = new THREE.SphereGeometry(0.12, 8, 8);
    const material = new THREE.MeshBasicMaterial({
      color: colorHex,
      transparent: true,
      opacity: 0.9,
    });
    this.mesh = new THREE.Mesh(geometry, material);
    this.updatePosition(0);
  }

  updatePosition(t: number) {
    this.progress = t % 1;
    const pos = this.curve.getPointAt(this.progress);
    this.mesh.position.copy(pos);
  }

  animateTravel(duration = 2000, loop = true) {
    return animate(this, {
      progress: 1,
      duration: duration,
      ease: 'inOut(2)',
      loop: loop,
      onUpdate: () => {
        this.updatePosition(this.progress);
      },
    });
  }
}
```

## 3. Ambient Volumetric Particle Dust (THREE.Points)

To render 400 floating embers around the 3D nucleus with zero performance impact, we use a single `THREE.Points` object with GPU vertex shader size attenuation:

```typescript
import * as THREE from 'three';

export function createAmbientDustField(count = 350, radius = 22): THREE.Points {
  const geometry = new THREE.BufferGeometry();
  const positions = new Float32Array(count * 3);
  const randomOffsets = new Float32Array(count * 3);

  for (let i = 0; i < count; i++) {
    // Spherical distribution
    const r = radius * Math.cbrt(Math.random());
    const theta = Math.random() * 2 * Math.PI;
    const phi = Math.acos(2 * Math.random() - 1);

    positions[i * 3 + 0] = r * Math.sin(phi) * Math.cos(theta);
    positions[i * 3 + 1] = r * Math.sin(phi) * Math.sin(theta);
    positions[i * 3 + 2] = r * Math.cos(phi);

    // Random drift frequencies for vertex shader
    randomOffsets[i * 3 + 0] = Math.random() * 10;
    randomOffsets[i * 3 + 1] = Math.random() * 10;
    randomOffsets[i * 3 + 2] = Math.random() * 10;
  }

  geometry.setAttribute('position', new THREE.BufferAttribute(positions, 3));
  geometry.setAttribute('aOffset', new THREE.BufferAttribute(randomOffsets, 3));

  const material = new THREE.ShaderMaterial({
    uniforms: {
      uTime: { value: 0 },
      uColor: { value: new THREE.Color('#D1D1D6') },
    },
    vertexShader: `
      uniform float uTime;
      attribute vec3 aOffset;
      varying float vAlpha;

      void main() {
        vec3 pos = position;
        // Subtle Brownian sinusoidal drift
        pos.x += sin(uTime * 0.4 + aOffset.x) * 0.4;
        pos.y += cos(uTime * 0.3 + aOffset.y) * 0.4;
        pos.z += sin(uTime * 0.5 + aOffset.z) * 0.4;

        vec4 mvPosition = modelViewMatrix * vec4(pos, 1.0);
        gl_Position = projectionMatrix * mvPosition;

        // Size attenuation based on distance to camera
        gl_PointSize = 2.5 * (200.0 / -mvPosition.z);
        vAlpha = 0.3 + 0.3 * sin(uTime * 1.5 + aOffset.x);
      }
    `,
    fragmentShader: `
      uniform vec3 uColor;
      varying float vAlpha;

      void main() {
        // Render soft circular anti-aliased particle
        float dist = length(gl_PointCoord - vec2(0.5));
        if (dist > 0.5) discard;
        float strength = 1.0 - smoothstep(0.0, 0.5, dist);

        gl_FragColor = vec4(uColor, strength * vAlpha);
      }
    `,
    transparent: true,
    depthWrite: false,
  });

  return new THREE.Points(geometry, material);
}
```

## 4. Multi-Edge Cascade Signal Burst

When an architectural node is triggered, an energy wave fans out into multiple outgoing edges simultaneously:

```typescript
import { createTimeline, stagger } from 'animejs';

export function fireTelemetrySignalBurst(photons: SplinePhoton[]) {
  const tl = createTimeline();

  photons.forEach((photon, index) => {
    photon.progress = 0;
    tl.add(
      photon,
      {
        progress: 1,
        duration: 900,
        ease: 'out(3)',
        onUpdate: () => photon.updatePosition(photon.progress),
      },
      index * 60 // 60ms stagger across branches
    );
  });

  return tl;
}
```
